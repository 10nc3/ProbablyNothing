const PHI = 1.618;

/**
 * Measure affordability for a city
 * @param {Object} params
 * @param {string} params.city - City name
 * @param {number} params.year - Year to analyze
 * @param {number} params.landPricePerSqm - Price per m² (USD)
 * @param {number} params.medianIncome - Median HH income (USD/year)
 * @returns {Object} Affordability analysis
 */
function measureAffordability({ city, year, landPricePerSqm, medianIncome }) {
  const LAND_QUANTA = 700; // m²/HH
  
  const totalPrice = landPricePerSqm * LAND_QUANTA;
  const ratio = totalPrice / medianIncome;
  const yearsToMortgage = ratio; // Simplified
  
  let regime, sigma;
  
  if (yearsToMortgage > 25 || ratio > 3.5) {
    regime = "FATALISM";
    sigma = (ratio - 3.5) / 3.5; // Substrate stress
  } else if (yearsToMortgage < 10) {
    regime = "OPTIMISM";
    sigma = -(10 - yearsToMortgage) / 10; // Substrate abundance
  } else {
    regime = "φ-BREATHING";
    sigma = 0; // Near equilibrium
  }
  
  // Solve A = 1 + 1/A + σ
  const A = solveIdentity(sigma);
  
  return {
    city,
    year,
    landPricePerSqm,
    medianIncome,
    totalPrice,
    ratio,
    yearsToMortgage,
    regime,
    sigma,
    identityValue: A,
    phiDeviation: Math.abs(A - PHI),
    metadata: {
      landQuanta: LAND_QUANTA,
      threshold: { fatalism: 3.5, optimism: 10/3.5 },
      fertilityWindow: 25
    }
  };
}

/**
 * Solve A = 1 + 1/A + σ
 */
function solveIdentity(sigma = 0) {
  // A² - A(1 + σ) - 1 = 0
  const a = 1;
  const b = -(1 + sigma);
  const c = -1;
  
  const discriminant = b*b - 4*a*c;
  if (discriminant < 0) {
    throw new Error(`No real solution for σ=${sigma}`);
  }
  
  const A_pos = (-b + Math.sqrt(discriminant)) / (2*a);
  const A_neg = (-b - Math.sqrt(discriminant)) / (2*a);
  
  return A_pos > 0 ? A_pos : A_neg;
}

/**
 * Compare two time periods for a city
 */
function compareTimePeriods(measurement1, measurement2) {
  const deltaRatio = measurement2.ratio - measurement1.ratio;
  const deltaYears = measurement2.yearsToMortgage - measurement1.yearsToMortgage;
  const regimeChange = measurement1.regime !== measurement2.regime;
  
  return {
    city: measurement1.city,
    period: `${measurement1.year} → ${measurement2.year}`,
    deltaRatio: deltaRatio.toFixed(2),
    deltaYears: deltaYears.toFixed(1),
    regimeChange: regimeChange ? `${measurement1.regime} → ${measurement2.regime}` : 'stable',
    direction: deltaRatio > 0 ? 'WORSENING' : 'IMPROVING',
    measurements: [measurement1, measurement2]
  };
}

/**
 * Detect seed metric intent in query
 */
function detectSeedMetricIntent(text) {
  if (!text) return false;
  
  const patterns = [
    /\b(700\s*m²|700\s*square\s*meters?)\b/i,
    /\b(affordability|affordable|price\s*to\s*income|housing\s*crisis)\b/i,
    /\b(fatalism|optimism|phi[\s\-]?breathing)\b/i,
    /\b(land\s*quanta|substrate\s*metric)\b/i,
    /\b(fertility\s*window|25\s*years?)\b/i,
    /\b(seed\s*metric|human\s*substrate)\b/i,
  ];
  
  return patterns.some(p => p.test(text));
}
